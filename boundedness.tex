\section{Boundedness Analysis}\label{sec:boundedness}

This section presents the actual boundedness analysis, that uses
the results of a preliminary side-effect analysis (Sec.~\ref{sec:side_effect}).

\emph{Paths} are expressions of the language
(Def.~\ref{def:exp_com}) that start from a variable and dereferences \<final> fields,
hence their value cannot be affected by side-effect.
%
\begin{definition}[Paths]\label{def:path}
  The set $\paths\subseteq\Exp$ of \emph{paths} is the minimal set such that
  \begin{itemize}
  \item if $v\in\Var$ then $v\in\paths$ is rooted at $v$ and its type is the declaration type of $v$;
  \item if $\pi\in\paths$ is rooted at $v$ and
    has type $C$ and if $C.f$ is a \<final> field of type $t$ defined in class $C$,
    then $\pi.f\in\paths$ is rooted at $v$ and has type $t$.
  \end{itemize}
\end{definition}
%
\begin{example}
  At line~\ref{ponzi1:require} in Fig.~\ref{fig:ponzi1},
  \<this> is a path rooted at \<this> of type \<Ponzi1>,
  \<amount> is a path rooted at \<amount> of type \<int> and
  \<this.investors> is a path rooted at \<this> of type \<List<Contract\texttt{>}>.
\end{example}
%
A path $\pi$ bound to a list $l$ cannot see $l$ replaced by side-effect,
but can see $l$ modified by side-effect (for instance, enlarged).
Otherwise, $\pi$ is said to \emph{survive}.
%
\begin{definition}[Path Survival]\label{def:survival}
  Given $\pi\in\paths$ and a set of types $U$, the path $\pi$ is said to
  \emph{survive any update to a list of a type in $U$} (or just
  \emph{$\pi$ survives $U$}) if
  \begin{itemize}
  \item $\pi$'s type is not a list, or
  \item $\pi$'s type is \<List<$t'$\texttt{>}> and $t'$ has no shared subtype with any $t''\in U$.
  \end{itemize}
\end{definition}

\begin{example}
  In Fig.~\ref{fig:ponzi1}, the path \<this.investors> has type \<List<Contract\texttt{>}>
  and hence it does not survive $\{\<Contract>\}$ (just take $t'=t''=\<Contract>$
  in Def.~\ref{def:survival}).
  The path \<amount> has type \<int>, hence it survives $\{\<Contract>\}$.
  In Fig.~\ref{fig:lottery1}, at line~\ref{lottery1:expand}, the path
  \<this.rounds> has type \<List<Round\texttt{>}> hence it survives
  $\{\<TicketBatchSale>\}$, since \<Contract> and \<TicketBatchSale> have no
  shared subtype. But \<List<Round\texttt{>}> does not survive $\{\<Object>\}$ nor
  $\{\<TicketBatchSale>,\<Object>\}$: just take $t'=\<Round>$ and $t''=\<Object>$
  and observe that \<Contract> is subtype of \<Object>.
\end{example}

\begin{definition}[Boundedness States]\label{def:bo}
  A \emph{boundedness state} \|bo| is
  \[
  \left\{\left.\begin{array}{c}
  \size{\pi_1}\le n_1\\
  \vdots\\
  \size{\pi_k}\le n_k
  \end{array}\right|\pi_i\in\paths\text{ and }n_i\in\nat\text{ for every $i=1,\ldots,k$}\right\}.
  \]
  The set of all possible boundedness states is $\botwo$.
\end{definition}
%
The intuition is that \|bo| represents the set of all concrete states
where $\pi_i$ holds a list whose length is at most $n_i$, for every $i=1,\ldots,k$.
%
\begin{proposition}[Abstract Domain]\label{prop:abstract_domain}
  In terms of abstract interpretation, any $\|bo|\in\botwo$ is an abstraction
  of concrete states and $\botwo$ is an abstract domain whose elements
  are ordered as $\|bo|_1\sqsubseteq\|bo|_2$
  if and only if for every $(\size{\pi}\le n)\in\|bo|_2$ there exists
  $(\size{\pi}\le n')\in\|bo|_1$ with $n'\le n$.
  Therefore, $\varnothing$ is the top (least precise) element of $\botwo$.
  The least upper bound operator on $\botwo$ is
  \[
  \|bo|_1\sqcup\|bo|_2=\left\{\size{\pi}\le\max(n_1,n_2)\left|\begin{array}{l}
  (\size{\pi}\le n_1)\in\|bo|_1\\
  (\size{\pi}\le n_2)\in\|bo|_2
  \end{array}\right.\right\}.
  \]
\end{proposition}
%
An assignment can invalidate some bounds, but only if $\pi$
is a single variable, since paths only refer to \<final> fields,
whose value cannot be changed.
%
\begin{definition}[Bounds Invalidation]\label{def:invalidation}
  Given $\|bo|\in\botwo$ and $\pi\in\paths$, the \emph{invalidation of $\pi$ in $\|bo|$} is
  \[
  \|bo|^{-\pi}=\begin{cases}
  \{(\size{\pi'}\le n)\in\|bo|\mid\pi'\text{ is not rooted at }v\} & \text{if $\pi$ is a variable $v$}\\
  \|bo| & \text{otherwise.}
  \end{cases}
  \]
\end{definition}
%
A comparison operation between two expressions can improve a boundedness state,
when it carries information about the size of a path.
Similarly, an assignment can sometimes improve a boundedness state.
%
\begin{definition}[Bounds Improvement and Assignment]\label{def:improvement}
  Given $\|bo|\in\botwo$ and a comparison expression $\cop{\exp_1}{\exp_2}$, the
  \emph{improvement of \|bo| \wrt $\cop{\exp_1}{\exp_2}$} is
  \begin{align*}
    \|improve|(\|bo|,\copfull{\size{\pi}}{\le}{n})&=\|bo|\cup\{\size{\pi}\le n\}\\
    \|improve|(\|bo|,\copfull{\size{\pi}}{<}{n})&=\|bo|\cup\{\size{\pi}\le n-1\}\\
    \|improve|(\|bo|,\copfull{n}{\ge}{\size{\pi}})&=\|bo|\cup\{\size{\pi}\le n\}\\
    \|improve|(\|bo|,\copfull{n}{>}{\size{\pi}})&=\|bo|\cup\{\size{\pi}\le n-1\}\\
    \|improve|(\|bo|,\cop{\exp_1}{\exp_2})&=\|bo|\text{ in all other cases.}
  \end{align*}
  The \emph{assignment in \|bo|} is ($\pi,\pi'\in\paths$):
  \begin{align*}
    \|assign|(\|bo|,\assign{\pi}{\new{\mathtt{List}}{}})&=\|bo|^{-\pi}\cup\{\size{\pi}\le 0\}\\
    \|assign|(\|bo|,\assign{\pi}{\pi'})&=\|bo|^{-\pi}\cup\{\size{\pi}\le n\mid(\size{\pi'}\le n)\in\|bo|\}\\
    \|assign|(\|bo|,\assign{\pi}{\exp})&=\|bo|^{-\pi}\text{ in all other cases not considered above}\\
    \|assign|(\|bo|,\assign{\exp}{\exp'})&=\|bo|\text{ if $\exp\not\in\paths$.}
  \end{align*}
\end{definition}

Boundedness is a property of computation states hence we present
its analysis in operational form, as a derivation rule over the boundedness states
of Def.~\ref{def:bo}.
%

  %
  The boundedness derivation rule $\boderivation$ for expressions
  in Fig.~\ref{fig:bo_derivations_expressions} is a big step semantics
  $\left(\Exp\times\botwo\right)\boderivation\botwo$
  that derives the approximation of the post-states
  from the approximation of the pre-states.
  The boundedness derivation rule $\boderivation$ for commands
  in Fig.~\ref{fig:bo_derivations_commands1} and~\ref{fig:bo_derivations_commands2}
  is a small step semantics
  $\left(\Com\times\botwo\right)\boderivation(\Com\cup\{\varepsilon\}\times\botwo)$.
  A derivation $\pair{\com_1}{\|bo|}\boderivation\pair{\com_2}{\|bo|'}$ means that
  the execution of $\com_1$ from a boundedness state \|bo| performs a single
  step and then continues with the execution of $\com_2$ from a boundedness state $\|bo|'$.
  A configuration $\pair{\varepsilon}{\|bo|}$ is a final configuration, reached when
  the execution stops.
  %

      \frac{\pair{\exp_1}{\|bo|}\boderivation\|bo|'\quad\pair{\exp_2}{\|bo|'}\boderivation\|bo|''}{\pair{\assign{\field{\exp_1}{f}}{\exp_2}\,\xi}{\|bo|}\boderivation\pair{\xi}{\|assign|(\|bo|'',\assign{\field{\exp_1}{f}}{\exp_2})}}\\[1pt]

      \frac{\pair{\exp_1}{\|bo|}\boderivation\|bo|'\quad\pair{\exp_2}{\|bo|'}\boderivation\|bo|''\quad\pair{\exp_3}{\|bo|''}\boderivation\|bo|'''}{\pair{\readmap{\exp_1}{\exp_2}{\exp_3}\,\xi}{\|bo|}\boderivation\pair{\xi}{\|bo|'''}}\\[1pt]

      \frac{\begin{array}{c}

      \frac{\pair{\exp_1}{\|bo|}\boderivation\|bo|'\quad\pair{\exp_2}{\|bo|'}\boderivation\|bo|''}{\pair{\receive{\exp_1}{\exp_2}\,\xi}{\|bo|}\boderivation\pair{\xi}{\|bo|''}}\\[1pt]

      \frac{\pair{\exp_1}{\|bo|}\boderivation\|bo|'\quad\pair{\exp_2}{\|bo|'}\boderivation\|bo|''}{\pair{\require{\exp_1}{\exp_2}\,\xi}{\|bo|}\boderivation{\pair{\xi}{\|improve|(\|bo|'',\cop{\exp_1}{\exp_2})}}}

      \pair{\sequence{\com_1}{\com_2}\,\xi}{\|bo|}\boderivation\pair{\com_1\,\com_2\,\xi}{\|bo|}\qquad

    \end{gather*}

    \label{fig:bo_derivations_commands2}
  \end{figure}
\end{definition}
%
In Fig.~\ref{fig:bo_derivations_expressions}, the
rule for constructor call first evaluates
the actual parameters, recursively, then projects the result on the bounds
that are not affected by the side-effects of the constructor. This
rule is intraprocedural, since it does not enter the body of the constructor
but simply embeds its side-effects in the result.
In Fig.~\ref{fig:bo_derivations_commands1}, the rule for method call
works similarly, but considers the
union of the side-effects of all targets of the call. The rules for assignment
update the boundness aproximation with the effects of the assignment
(Def.~\ref{def:improvement}). The rule for \<push> is the most interesting.
It evaluates receiver and parameter of \<push> and then splits the
resulting bounds into those about lists whose static type guarantees that
they are unaffected by the \<push> ($\|bo|_{\mathit{survives}}$) and those for lists
that, instead,
might be affected ($\|bo|_{\neg\mathit{survives}}$). The approximation of the size
of the former remains valid, while that of the latter is conservatively
weakened by a unit (one element might be added to those lists).
The rule for \<require> tries to improve (Def.~\ref{def:improvement}) the boundedness state
with information entailed by the required condition, that must hold or
otherwise the computation stops. In Fig.~\ref{fig:bo_derivations_commands2},
the same occurs for the conditional, that non-deterministically proceeds
to the \emph{then} branch if the condition holds and to the \emph{else} branch
if its opposite holds. Finally, the rule for $\for{v}{\exp}{\com}$ evaluates
$\exp$ and then executes a command $\com^{*,v}$ whose semantics
is to execute $\com$ zero or more times, non-deterministically,
every time reassigning $v$ before each iteration.

The derivation rule in Fig.~\ref{fig:bo_derivations_commands1} and
in~\ref{fig:bo_derivations_commands2} has been given
as a small step semantics in order to allow the computation of a
boundedness state at the beginning of every command in the program, through an
intraprocedural fixpoint that saturates all execution paths. Namely, at the beginning
of each method with body $\com$, the analysis starts from the configuration
$\pair{\com}{\varnothing}$ ($\varnothing$ is a worst-case assumption at the beginning
of the method: nothing is known there about the size of the lists)
and computes all possible derivations from there.
Whenever a configuration $\pair{\com'\,\xi}{\|bo|}$ is reached, the analysis records
$\|bo|$ as the state at the program point of $\com'$, if no recording exists there yet.
If, instead, a previous recorded state $\|bo|_{\|old|}$ exists there,
the analysis records $\|bo|_{\|new|}=\|bo|_{\|old|}\sqcup\|bo|$ as the state at that program point.
If $\|bo|_{\|new|}\not=\|bo|_{\|old|}$, the derivation continues from $\|bo|_{\|new|}$.
Otherwise, the derivation stops.

In Ex.~\ref{ex:boundedness_computation}, the computation of the boundedness states
converges. This is expected for simple code such as smart contracts. However,
$\botwo$ has infinite ascending chains hence that computation
might diverge, in general, and a widening operator is needed, to force
termination after a given number $\kappa$ of iterations~\cite{CousotC77}.
%
\begin{definition}[Widening Operator]\label{def:widening}
  The \emph{widening operator on $\botwo$} is set intersection:
  $\nabla(\|bo|_1,\|bo|_2)=\|bo|_1\cap\|bo|_2$.
\end{definition}
%
By using a widening, when reaching a program point with an approximation $\|bo|$,
the new approximation at that program point is computed from the old one as
\[
\|bo|_{\|new|}=\begin{cases}
\|bo|_{\|old|}\sqcup\|bo| & \text{for the first $\kappa$ iterations}\\
(\|bo|_{\|old|}\sqcup\|bo|)\nabla\|bo|_{\|old|} & \text{from the $(\kappa+1)$th iteration.}\\
\end{cases}
\]
%
\begin{proposition}[Termination]\label{prop:termination}
  The operator $\nabla$ is a widening operator on finite elements of $\botwo$.
  Therefore, the computation
  of a boundedness state at each program point must terminate in
  a finite number of iterations.
\end{proposition}

In the following, we assume that a boundedness state has been computed at each program point.
This information is used to warn about boundedness bugs.
%
\begin{definition}[Bounded Lists]\label{def:bounded_lists}
  All lists of type $t$ are \emph{bounded} in a program
  if for every $\push{\exp_1}{\exp_2}$ in the program such that
  $\<List<$t'$\texttt{>}>$ is the type of $\exp_1$, 
  $t$ and $t'$ share a subtype and $\|bo|$ is the boundedness state computed
  at the \<push> command,
  it is $\exp_1=\pi\in\paths$ and $(\size{\pi}\le n)\in\|bo|$ for some $n\in\nat$.
\end{definition}
%
\begin{example}\label{ex:bounded_lists}
  Not all lists of $t=\<Contract>$ are bounded
  in \<Ponzi1> (Fig.~\ref{fig:ponzi1}), since the \<push>
  at line~\ref{ponzi1:expand} operates on $\<this.investors>$, of type
  $\<List<$t'$\texttt{>}>$ with $t'=\<Contract>$, $t$ and $t'$ share a subtype
  (\ieinitial \<Contract> itself), $\<this.investors>\in\paths$
  but the boundedness state computed at that \<push> is
  $\varnothing$ (Fig.~\ref{fig:boundedness_ponzi1}) and no
  $(\size{\mathtt{this.investors}}\le n)\in\varnothing$, for any $n\in\nat$.
  The same holds for \<Ponzi2> (Fig.~\ref{fig:ponzi2}).
  Instead, in \<Ponzi3> (Fig.~\ref{fig:ponzi3}), all lists of type $\<Contract>$
  are bounded since the boundedness state computed at the \<push>
  at line~\ref{ponzi3:creator} is $\{\size{\mathtt{this.investors}}\le 0\}$
  (Fig.~\ref{fig:boundedness_ponzi3})
  and that computed at the \<push> at line~\ref{ponzi3:expand} is
  $\{\size{\mathtt{this.investors}}\le 99\}$ (Fig.~\ref{fig:boundedness_ponzi3}).
  Similarly, in \<Lottery1> (Fig.~\ref{fig:lottery1}),
  not all lists of \<Round> and not all lists of \<TicketBatchSale> are bounded,
  since the boundedness state computed at the
  \<push> commands inside \<buy()> (Fig.~\ref{fig:lottery1}),
  that operate on such lists, is $\varnothing$ (Fig.~\ref{fig:boundedness_lottery1}).
  For the same reason, in \<Lottery2> (Fig.~\ref{fig:lottery2}),
  not all lists of \<Round> are bounded but all lists of \<TicketBatchSale>
  are bounded, since the only \<push> that operates on the latter
  is in \<buy()>, it works on $\<last.sale>\in\paths$ and
  the boundedness state there is $\{\size{\mathtt{last.sale}}\le 4999\}$
  (Fig.~\ref{fig:boundedness_lottery2}). 
\end{example}

\begin{definition}[Warning Generation]\label{def:warnings}
  Our boundedness analyzer issues a warning at each
  $\for{v}{\exp}{\com}$ command in the program, where
  $\exp$ has type $\<List<$t$\texttt{>}>$, if and only if it is not true that
  all lists of type $t$ are bounded (Def.~\ref{def:bounded_lists}).
\end{definition}
%
\begin{example}\label{ex:warnings}
  Our analyzer warns ar line~\ref{ponzi1:iteration} of
  \<Ponzi1> (Fig.~\ref{fig:ponzi1}), where a \<for> operates on
  \<this.investors> of type \<List<Contract\texttt{>}> but not all lists of
  \<Contract> are bounded in \<Ponzi1> (Ex.~\ref{ex:bounded_lists}).
  Similarly, it warns at line~\ref{ponzi2:iteration} of \<Ponzi2>
  (Fig.~\ref{fig:ponzi2}). But it does not warn
  in \<Ponzi3> (Fig.~\ref{fig:ponzi3}), since all lists of \<Contract> are bounded there
  (Ex.~\ref{ex:bounded_lists}). It warns instead
  at line~\ref{lottery1:iteration} of \<Lottery1> (Fig.~\ref{fig:lottery1}),
  since the \<for> command there operates on \<this.round.sales> of type
  \<List<TicketBatchSale\texttt{>}> but not all lists of
  \<TicketBatchSale> are bounded in \<Lottery1> (Ex.~\ref{ex:bounded_lists}).
  It does not warn at that same \<for> command in \<Lottery2>
  (Fig.~\ref{fig:lottery2}), since all lists of \<TicketBatchSale> are bounded there
  (Ex.~\ref{ex:bounded_lists}). These results are exactly
  those foreseen in Sec.~\ref{sec:examples}.
\end{example}
