\section{Boundedness Analysis}\label{sec:boundedness}

The \emph{paths} are a subset of the expressions of the language
(Def.~\ref{def:exp_com}) that starts from a variable and dereferences \<final> fields.
The limitation to \<final> fields simplifies the analysis, since the value of a path
cannot be affected by side-effect.
%
\begin{definition}[Paths]\label{def:path}
  The set $\paths{}\subseteq\Exp$ of \emph{paths} is the minimal set such that
  \begin{itemize}
  \item if $v$ is a local variable,
    then $v\in\paths{}$ is rooted at $v$ and its type is the declaration type of $v$
    (all local variables are a path);
  \item if $\pi\in\paths{}$ is rooted at $v$ and
    has type $C$ and if $\mathtt{C.f}$ is a \<final> field
    of type $t$ defined in class $C$,
    then $\pi.f\in\paths{}$ is rooted at $v$ and its type is $t$
    (paths can be expanded by \<final> field dereference).
  \end{itemize}
\end{definition}
%
\begin{example}
  At line~\ref{ponzi1:require} in Fig.~\ref{fig:ponzi1},
  \<this> is a path rooted at \<this> of type \<Ponzi1>,
  \<amount> is a path rooted at \<amount> of type \<int> and
  \<this.investors> is a path rooted at \<this> of type \<List<Contract\texttt{>}>.
\end{example}

The following definition specifies when a list bound to a path (if any)
is not affected by the execution of some piece of code that might update some lists.
We will use this is a moment in conjunction with the side-effects computed in
Sec.~\ref{sec:side_effect}.
%
\begin{definition}[Path Survival]\label{def:survival}
  Given $\pi\in\paths{}$ and a set of types $U$, the path $\pi$ is said to
  \emph{survive any update to a list of a type in $U$} (or just
  \emph{$\pi$ survives $U$}) if
  \begin{itemize}
  \item $\pi$'s type is not a list, or
  \item $\pi$'s type is \<List<$t'$\texttt{>}> and $t'$
    has no shared subtype with any $t''\in S$.
  \end{itemize}
\end{definition}

\begin{example}
  In Fig.~\ref{fig:ponzi1}, the path \<this.investors> has type \<List<Contract\texttt{>}>
  and hence it does not survive $\{\<Contract>\}$ (just take $t'=t''=\<Contract>$
  in Def.~\ref{def:survival}).
  The path \<amount> has type \<int>, hence it survives $\{\<Contract>\}$.
  In Fig.~\ref{fig:lottery1}, at line~\ref{lottery1:expand}, the path
  \<this.rounds> has type \<List<Round\texttt{>}> hence it survives
  $\{\<TicketBatchSale>\}$, since \<Contract> and \<TicketBatchSale> have no
  shared subtype. But it does not survive $\{\<Object>\}$ nor
  $\{\<TicketBatchSale>,\<Object>\}$: just take $t'=\<Round>$ and $t''=\<Object>$
  and observe that \<Contract> is subtype of \<Object>.
\end{example}

\begin{definition}[Boundedness States]\label{def:bo}
  A \emph{boundedness state} $\|bo|$ is a set of constraints
  \[
  \|bo|=\left\{\left.\begin{array}{c}
  \size{\pi_1}\le n_1\\
  \ldots\\
  \size{\pi_k}\le n_k
  \end{array}\right|\pi_i\in\paths{}\text{ and }n_i\in\nat\text{ for every $i=1,\ldots,k$}\right\}.
  \]
  The set of all possible boundedness states is $\botwo$.
\end{definition}
%
The intuition is that \|bo| represents the set of all concrete states
where $\pi_i$ holds a list whose length is at most $n_i$, for every
$i=1,\ldots,k$. In terms of abstract interpretation, \|bo| is an abstraction
of sets of such concrete states and $\botwo$ is an abstract domain.

An assignment to a path invalidates somebounds, but only if it
is a single variable, since we are assuming that paths only refer to \<final> fields.
%
\begin{definition}[Bounds Invalidation]\label{def:invalidation}
  Given $\|bo|\in\botwo$ and $\pi\in\paths{}$, the \emph{invalidation of $\pi$ in $\|bo|$} is
  \[
  \|bo|^{-\pi}=\begin{cases}
  \{(\size{\pi}\le n)\in\|bo|\mid\pi\text{ is not rooted at }v\} & \text{if $\pi$ is a variable $v$}\\
  \|bo| & \text{otherwise.}
  \end{cases}
  \]
\end{definition}

A comparison operation between two expressions can improve a boundedness state,
when it carries information about a bound on the size of a path.
Similarly, an assignment can improve a boundedness state if it assigns a path.
%
\begin{definition}[Bounds Improvement]\label{def:improvement}
  Given $\|bo|\in\botwo$ and a comparison expression $\cop{\exp_1}{\exp_2}$, the
  \emph{improvement of \|bo| \wrt $\cop{\exp_1}{\exp_2}$} is
  \begin{align*}
    \|improve|(\|bo|,\copfull{\size{\pi}}{\le}{n})&=\|bo|\cup\{\size{\pi}\le n\}\\
    \|improve|(\|bo|,\copfull{\size{\pi}}{<}{n})&=\|bo|\cup\{\size{\pi}\le n-1\}\\
    \|improve|(\|bo|,\copfull{n}{\ge}{\size{\pi}})&=\|bo|\cup\{\size{\pi}\le n\}\\
    \|improve|(\|bo|,\copfull{n}{>}{\size{\pi}})&=\|bo|\cup\{\size{\pi}\le n-1\}\\
    \|improve|(\|bo|,\cop{\exp_1}{\exp_2})&=\|bo|\text{ in all other cases}
  \end{align*}
  The \emph{assignment to a path in \|bo|} is
  \begin{align*}
    \|assign|(\|bo|,\assign{\pi}{\new{\mathtt{List}}{}})&=\|bo|^{-\pi}\cup\{\size{\pi}\le 0\}\\
    \|assign|(\|bo|,\assign{\pi}{\pi'})&=\|bo|^{-\pi}\cup\{\size{\pi}\le n\mid(\size{\pi'}\le n)\in\|bo|\}\\
    \|assign|(\|bo|,\assign{\pi}{\exp})&=\|bo|^{-\pi}\text{ in all other cases}
  \end{align*}
  where $\pi,\pi'\in\paths{}$.
\end{definition}

Since boundedness is a property of computation states, boundedness analysis
will be presented in operational form, as a derivation rule over the abstract states
from Def.~\ref{def:bo}.
%
\begin{definition}[Boundedness Derivation Rules]\label{def:boundness_rules}
  The boundedness derivation rule $\boderivation$ for expressions
  in Fig.~\ref{fig:bo_derivations_expressions} is a big step semantics of the form
  \[
  \left(\Exp\times\botwo\right)\boderivation\botwo
  \]
  that derives the approximation of the post-states
  from the approximation of the pre-states.
  The boundedness derivation rule $\boderivation$ for commands
  in Fig.~\ref{fig:bo_derivations_commands} is a small step semantics of the form
  \[
  \left(\Com\times\botwo\right)\boderivation(\Com\cup\{\varepsilon\}\times\botwo).
  \]
  %
  \begin{figure}[t]
    \begin{gather*}
      \pair{c}{\|bo|}\boderivation\|bo|\qquad\pair{v}{\|bo|}\boderivation\|bo|\\[5pt]
      \frac{\begin{array}{c}
          \pair{\exp_1}{\|bo|}\boderivation\|bo|_1\quad\cdots\quad\pair{\exp_p}{\|bo|_{p-1}}\boderivation\|bo|_p\\
          \|bo|'=\{(\size{\pi}\le n)\in\|bo|_p\mid\pi\text{ survives }\intsideeffect(C.C)\}
      \end{array}}{\pair{\new{C}{\exp_1,\ldots,\exp_p}}{\|bo|}\boderivation\|bo|'}\\[5pt]
      \frac{\pair{\exp}{\|bo|}\boderivation\|bo|'}{\pair{\assign{v}{\exp}}{\|bo|}\boderivation\{(\size{\pi}\le n)\in\|bo|'\mid\pi\text{ is not rooted at }v\}}\\[5pt]
      \text{the rules for the other expressions work by recursive descent, as for instance below:}\\
      \frac{\pair{\exp}{\|bo|}\boderivation\|bo|'}{\pair{\field{\exp}{f}}{\|bo|}\boderivation\|bo|'}
    \end{gather*}
    \caption{The boundedness derivation rules for expressions. They use a previously computed side-effect analysis $\intsideeffect$.}
    \label{fig:bo_derivations_expressions}
  \end{figure}
%
  \begin{figure}[t]
    \begin{gather*}
      % variable assignment
      \frac{\pair{\exp}{\|bo|}\boderivation\|bo|'}{\pair{\assign{v}{\exp}\,\xi}{\|bo|}\boderivation\pair{\xi}{\|assign|(\|bo|',\assign{v}{\exp})}}\\[5pt]
      % field assignment
      \frac{\pair{\exp_1}{\|bo|}\boderivation\|bo|'\quad\pair{\exp_2}{\|bo|'}\boderivation\|bo|''}{\pair{\assign{\field{\exp_1}{f}}{\exp_2}\,\xi}{\|bo|}\boderivation\pair{\xi}{\|assign|(\|bo|'',\assign{\field{\exp_1}{f}}{\exp_2})}}\\[5pt]
      % map assignment
      \frac{\pair{\exp_1}{\|bo|}\boderivation\|bo|'\quad\pair{\exp_2}{\|bo|'}\boderivation\|bo|''\quad\pair{\exp_3}{\|bo|''}\boderivation\|bo|'''}{\pair{\readmap{\exp_1}{\exp_2}{\exp_3}\,\xi}{\|bo|}\boderivation\pair{\xi}{\|bo|'''}}\\[5pt]
      % push
      \frac{\begin{array}{c}
          \pair{\exp_1}{\|bo|}\boderivation\|bo|'\quad\pair{\exp_2}{\|bo|'}\boderivation\|bo|''\quad\exp_1\text{ has type \<List<$t$\texttt{>}>}\\
          \|bo|_{\text{survives}}=\{(\size{\pi}\le n)\in\|bo|''\mid\pi\text{ survives }\{t\}\}\\
          \|bo|_{\text{not survives}}=\{\size{\pi}\le n+1\mid(\size{\pi}\le n)\in\|bo|''\text{ and }\pi\text{ does not survive }\{t\}\}
      \end{array}}{\pair{\push{\exp_1}{\exp_2}\,\xi}{\|bo|}\boderivation\pair{\xi}{\|bo|_{\text{survives}}\cup\|bo|_{\text{not survives}}}}\\[5pt]
      % receive
      \frac{\pair{\exp_1}{\|bo|}\boderivation\|bo|'\quad\pair{\exp_2}{\|bo|'}\boderivation\|bo|''}{\pair{\receive{\exp_1}{\exp_2}\,\xi}{\|bo|}\boderivation\pair{\xi}{\|bo|''}}\\[5pt]
      % method call
      \frac{\begin{array}{c}
          \pair{\exp_0}{\|bo|}\boderivation\|bo|_0\quad\cdots\quad\pair{\exp_p}{\|bo|_{p-1}}\boderivation\|bo|_p\\
          \text{$C_1.m\cdots C_k.m$ are the methods possibly called here}\\
          \|bo|'=\{(\size{\pi}\le n)\in\|bo|_p\mid\pi\text{ survives }\cup_{i=1}^k\intsideeffect(C_j.m)\}
      \end{array}}{\pair{\call{\exp_0}{m}{\exp_1,\ldots,\exp_p}\,\xi}{\|bo|}\boderivation\pair{\xi}{\|bo|'}}\\[5pt]
      % require
      \frac{\pair{\exp_1}{\|bo|}\boderivation\|bo|'\quad\pair{\exp_2}{\|bo|'}\boderivation\|bo|''}{\pair{\require{\exp_1}{\exp_2}\,\xi}{\|bo|}\boderivation{\pair{\xi}{\|improve|(\|bo|'',\cop{\exp_1}{\exp_2})}}}\\[5pt]
      % if/then/else
      \frac{\pair{\exp_1}{\|bo|}\boderivation\|bo|'\quad\pair{\exp_2}{\|bo|'}\boderivation\|bo|''}{\pair{\ife{\exp_1}{\exp_2}{\com_1}{\com_2}\,\xi}{\|bo|}\boderivation\pair{\com_1\,\xi}{\|improve|(\|bo|'',\cop{\exp_1}{\exp_2})}}\\[5pt]
      \frac{\pair{\exp_1}{\|bo|}\boderivation\|bo|'\quad\pair{\exp_2}{\|bo|'}\boderivation\|bo|''}{\pair{\ife{\exp_1}{\exp_2}{\com_1}{\com_2}\,\xi}{\|bo|}\boderivation\pair{\com_2\,\xi}{\|improve|(\|bo|'',\copfull{\exp_1}{\neg\|cop|}{\exp_2})}}\\[5pt]
      % for each
      \frac{\pair{\exp}{\|bo|}\boderivation\|bo|'}{\pair{\for{v}{\exp}{\com}\,\xi}{\|bo|}\boderivation\pair{\com^{*,v}\,\xi}{\|bo|'}}\\[5pt]
      \pair{\com^{*,v}\,\xi}{\|bo|}\boderivation\pair{\xi}{\|bo|}\qquad\pair{\com^{*,v}\,\xi}{\|bo|}\boderivation\pair{\com\,\com^{*,v}\,\xi}{\|bo|^{-v}}\\[5pt]
      % sequence
      \pair{\sequence{\com_1}{\com_2}\,\xi}{\|bo|}\boderivation\pair{\com_1\,\com_2\,\xi}{\|bo|}\qquad
      % nop
      \pair{\nop\,\xi}{\|bo|}\boderivation\pair{\xi}{\|bo|}
    \end{gather*}
    \caption{The boundedness derivation rules for commands. They use
      a previously computed side-effect analysis $\intsideeffect$. $\xi$ is the continuation
      that must be executed after the leftmost command terminates. It can be $\xi=\varepsilon$.}
    \label{fig:bo_derivations_commands}
  \end{figure}
\end{definition}
%
For instance, $\pair{\com_1}{\|bo|}\boderivation\pair{\com_2}{\|bo|'}$ means that
the execution of $\com_1$ in a boundedness state \|bo| performs a single
step and then continues with the execution of $\com_2$ from a boundedness state $\|bo|'$.
A configuration $\pair{\varepsilon}{\|bo|}$ is a final configuration, reached when
the execution stops.

The derivation rule in Fig.~\ref{fig:bo_derivations_commands} has been given
as a small step semantics in order to allow the computation of a
boundedness state at the beginning of every command in the program, through an
intraprocedural fixpoint that saturates all execution paths. Namely, at the beginning
of each method with body $\com$, the analysis starts from the configuration
$\pair{\com}{\varnothing}$ and computes all possible derivations from there.
Whenever a configuration $\pair{\com'\,\xi}{\|bo|}$ is reached, the analysis records
$\|bo|$ as current approximation at the program point of $\com'$. If the propagation reaches
a program point twice, it
records the least upper bound of the state previously recorded
there and of the new reached state, and the derivation continues from that
least upper bound.
